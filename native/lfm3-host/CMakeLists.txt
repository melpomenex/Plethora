cmake_minimum_required(VERSION 3.24)
project(plethora_lfm3_host LANGUAGES C CXX)
include(FetchContent)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "Load optional accelerators without requiring their drivers" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU baseline" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "Select a compatible CPU backend at runtime" FORCE)
# GGML_METAL defaults on for Apple. CUDA is an optional normal CMake backend.
FetchContent_Declare(llama
  URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/ac8e1694d837b11ed0aab944a6a2eee34b9c4d80
  URL_HASH SHA256=ac2df86939de1b4edded47d3aa56bf81179f5ea807d75206763a24501967f7f3)
FetchContent_MakeAvailable(llama)
add_executable(lfm3-host main.cpp)
target_compile_features(lfm3-host PRIVATE cxx_std_17)
target_link_libraries(lfm3-host PRIVATE llama llama-common mtmd)
if(APPLE)
  set_target_properties(lfm3-host PROPERTIES BUILD_RPATH "@loader_path/lfm3-runtime" INSTALL_RPATH "@loader_path/lfm3-runtime;@loader_path/../Resources/bin/lfm3-runtime")
elseif(UNIX)
  set_target_properties(lfm3-host PROPERTIES BUILD_RPATH "$ORIGIN/lfm3-runtime" INSTALL_RPATH "$ORIGIN/lfm3-runtime")
endif()
install(TARGETS lfm3-host RUNTIME DESTINATION .)
